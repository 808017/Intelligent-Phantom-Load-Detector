void sendToMQTT() {
  if (!mqttClient.connected()) return;
  
  // Send all measurements
  mqttClient.publish(voltage_feed, String(receivedData.liveVoltage, 2).c_str());
  mqttClient.publish(current_feed, String(receivedData.current, 4).c_str());
  mqttClient.publish(power_feed, String(receivedData.power, 2).c_str());
  mqttClient.publish(power_factor_feed, String(receivedData.powerFactor, 3).c_str());
  mqttClient.publish(phantom_feed, String(receivedData.phantomVoltage, 2).c_str());
  mqttClient.publish(energy_feed, String(totalEnergyWh, 3).c_str());
  
  // Send system status
  String status = receivedData.isMainsOn ? "MAINS_ON" : 
                  (receivedData.phantomVoltage > 0.5 ? "PHANTOM_DETECTED" : "OFF");
  mqttClient.publish(status_feed, status.c_str());
  
  // Create comprehensive JSON payload with enhanced data
  DynamicJsonDocument doc(2048);
  doc["live_voltage"] = round(receivedData.liveVoltage * 100) / 100.0;
  doc["current"] = round(receivedData.current * 10000) / 10000.0;
  doc["power"] = round(receivedData.power * 100) / 100.0;
  doc["power_factor"] = round(receivedData.powerFactor * 1000) / 1000.0;
  doc["phantom_voltage"] = round(receivedData.phantomVoltage * 100) / 100.0;
  doc["total_energy_wh"] = round(totalEnergyWh * 1000) / 1000.0;
  doc["daily_energy_wh"] = round(dailyEnergyWh * 1000) / 1000.0;
  doc["is_mains_on"] = receivedData.isMainsOn;
  doc["sender_timestamp"] = receivedData.timestamp;
  doc["receiver_timestamp"] = millis();
  doc["data_age_ms"] = millis() - lastDataReceived;
  doc["wifi_rssi"] = WiFi.RSSI();
  doc["sender_signal"] = receivedData.signalStrength;
  doc["alert_active"] = buzzerState;
  doc["current_alert"] = currentAlert;
  doc["wifi_channel"] = WiFi.channel();
  doc["apparent_power"] = round(receivedData.liveVoltage * receivedData.current * 100) / 100.0;
  
  // Add statistics
  JsonObject stats_obj = doc.createNestedObject("statistics");
  stats_obj["max_voltage"] = round(stats.maxVoltage * 100) / 100.0;
  stats_obj["min_voltage"] = round(stats.minVoltage * 100) / 100.0;
  stats_obj["max_current"] = round(stats.maxCurrent * 10000) / 10000.0;
  stats_obj["max_power"] = round(stats.maxPower * 100) / 100.0;
  stats_obj["avg_voltage"] = round(stats.avgVoltage * 100) / 100.0;
  stats_obj["avg_current"] = round(stats.avgCurrent * 10000) / 10000.0;
  stats_obj["avg_power"] = round(stats.avgPower * 100) / 100.0;
  stats_obj["data_points"] = stats.dataPoints;
  stats_obj["successful_tx"] = stats.successfulTransmissions;
  stats_obj["failed_tx"] = stats.failedTransmissions;
  stats_obj["voltage_stability"] = round(dataQuality.voltageStability * 100) / 100.0;
  stats_obj["avg_signal"] = round(dataQuality.averageSignalStrength * 10) / 10.0;
  
  String jsonString;
  serializeJson(doc, jsonString);
  
  mqttClient.publish(statistics_feed, jsonString.c_str());
  
  Serial.print("Sent to MQTT: ");
  Serial.print(jsonString.length());
  Serial.println(" bytes");
}

void updateLCD() {
  lcd.clear();
  
  if (!dataValid) {
    // No fresh data available
    lcd.setCursor(0, 0);
    lcd.print("NO DATA FROM    ");
    lcd.setCursor(0, 1);
    lcd.print("SENDER UNIT     ");
    lcd.setCursor(0, 2);
    lcd.print("Timeouts: ");
    lcd.print(dataQuality.consecutiveTimeouts);
    lcd.setCursor(0, 3);
    lcd.print("Ch:");
    lcd.print(WiFi.channel());
    lcd.print(" Age:");
    lcd.print((millis() - lastDataReceived) / 1000);
    lcd.print("s");
    return;
  }
  
  switch (displayMode) {
    case BASIC_DISPLAY:
      displayBasicInfo();
      break;
    case POWER_DISPLAY:
      displayPowerInfo();
      break;
    case ENERGY_DISPLAY:
      displayEnergyInfo();
      break;
    case STATUS_DISPLAY:
      displayStatusInfo();
      break;
    case STATISTICS_DISPLAY:
      displayStatistics();
      break;
    case DEBUG_DISPLAY:
      displayDebugInfo();
      break;
  }
  
  // Always show connection status on line 4
  displayConnectionStatus();
}

void displayBasicInfo() {
  // Line 1: Voltage
  lcd.setCursor(0, 0);
  lcd.print("Voltage: ");
  if (receivedData.liveVoltage > 10) {
    lcd.print((int)receivedData.liveVoltage);
    lcd.print("V");
  } else {
    lcd.print("OFF");
  }
  
  // Line 2: Current
  lcd.setCursor(0, 1);
  lcd.print("Current: ");
  if (receivedData.current >= 1.0) {
    lcd.print(receivedData.current, 2);
    lcd.print("A");
  } else if (receivedData.current >= 0.001) {
    lcd.print((int)(receivedData.current * 1000));
    lcd.print("mA");
  } else {
    lcd.print("0mA");
  }
  
  // Line 3: Power with alert indicator
  lcd.setCursor(0, 2);
  lcd.print("Power: ");
  if (receivedData.power >= 1000) {
    lcd.print(receivedData.power/1000, 1);
    lcd.print("kW");
  } else if (receivedData.power >= 1) {
    lcd.print((int)receivedData.power);
    lcd.print("W");
  } else {
    lcd.print("0W");
  }
  
  if (buzzerState) {
    lcd.setCursor(12, 2);
    lcd.print("ALRT");
  }
}

void displayPowerInfo() {
  // Line 1: Real Power
  lcd.setCursor(0, 0);
  lcd.print("Real: ");
  if (receivedData.power >= 1000) {
    lcd.print(receivedData.power/1000, 1);
    lcd.print("kW");
  } else {
    lcd.print((int)receivedData.power);
    lcd.print("W");
  }
  
  // Line 2: Apparent Power
  lcd.setCursor(0, 1);
  lcd.print("Apparent: ");
  float apparentPower = receivedData.liveVoltage * receivedData.current;
  if (apparentPower >= 1000) {
    lcd.print(apparentPower/1000, 1);
    lcd.print("kVA");
  } else {
    lcd.print((int)apparentPower);
    lcd.print("VA");
  }
  
  // Line 3: Power Factor with quality indicator
  lcd.setCursor(0, 2);
  lcd.print("PF: ");
  lcd.print(receivedData.powerFactor, 2);
  
  if (receivedData.powerFactor < 0.8 && receivedData.power > 50) {
    lcd.setCursor(12, 2);
    lcd.print("LOW");
  } else if (receivedData.powerFactor > 0.95 && receivedData.power > 50) {
    lcd.setCursor(12, 2);
    lcd.print("GOOD");
  }
}

void displayStatusInfo() {
  // Line 1: System Status
  lcd.setCursor(0, 0);
  if (receivedData.isMainsOn && receivedData.current > 0.001) {
    lcd.print("Status: LOAD ON ");
  } else if (receivedData.isMainsOn) {
    lcd.print("Status: MAINS ON");
  } else if (receivedData.phantomVoltage > 0.1) {
    lcd.print("Status: PHANTOM ");
  } else {
    lcd.print("Status: OFF       ");
  }
  
  // Line 2: Phantom voltage/current if present
  lcd.setCursor(0, 1);
  if (receivedData.phantomVoltage > 0.1) {
    lcd.print("Phantom: ");
    lcd.print((int)receivedData.phantomVoltage);
    lcd.print("V");
  } else {
    lcd.print("No Phantom       ");
  }
  
  // Line 3: Current alert
  lcd.setCursor(0, 2);
  if (currentAlert != "") {
    lcd.print("Alert: ");
    lcd.print(currentAlert);
    // Pad with spaces to clear line
    for (int i = 7 + currentAlert.length(); i < 16; i++) {
      lcd.print(" ");
    }
  } else {
    lcd.print("All Systems OK  ");
  }
}

void checkAlerts() {
  bool shouldAlert = false;
  String alertMsg = "";
  
  // Check for low voltage
  if (receivedData.isMainsOn && receivedData.liveVoltage < lowVoltageThreshold) {
    alertMsg = "LOW_VOLTAGE:" + String(receivedData.liveVoltage, 1) + "V";
    currentAlert = "LOW_V";
    shouldAlert = true;
    buzzerTone = 500; // Low frequency for low voltage
  }
  
  // Check for high voltage
  else if (receivedData.isMainsOn && receivedData.liveVoltage > highVoltageThreshold) {
    alertMsg = "HIGH_VOLTAGE:" + String(receivedData.liveVoltage, 1) + "V";
    currentAlert = "HIGH_V";
    shouldAlert = true;
    buzzerTone = 2000; // High frequency for high voltage
  }
  
  // Check for high current
  else if (receivedData.isMainsOn && receivedData.current > highCurrentThreshold) {
    alertMsg = "HIGH_CURRENT:" + String(receivedData.current, 2) + "A";
    currentAlert = "HIGH_I";
    shouldAlert = true;
    buzzerTone = 1800; // High frequency for high current
  }
  
  // Check for high power
  else if (receivedData.isMainsOn && receivedData.power > highPowerThreshold) {
    alertMsg = "HIGH_POWER:" + String(receivedData.power, 1) + "W";
    currentAlert = "HIGH_P";
    shouldAlert = true;
    buzzerTone = 1600; // Medium-high frequency for high power
  }
  
  // Check for phantom voltage
  else if (!receivedData.isMainsOn && receivedData.phantomVoltage > phantomThreshold) {
    alertMsg = "PHANTOM_DETECTED:" + String(receivedData.phantomVoltage, 1) + "V";
    currentAlert = "PHANTOM";
    shouldAlert = true;
    buzzerTone = 1500; // Medium frequency for phantom
  }
  
  // Check for poor power factor
  else if (receivedData.isMainsOn && receivedData.power > 100 && receivedData.powerFactor < lowPowerFactorThreshold) {
    alertMsg = "LOW_POWER_FACTOR:" + String(receivedData.powerFactor, 2);
    currentAlert = "LOW_PF";
    shouldAlert = true;
    buzzerTone = 800; // Low-medium frequency for power factor
  }
  
  // Check for data quality issues
  else if (dataQuality.voltageStability > 5.0 && receivedData.isMainsOn) {
    alertMsg = "VOLTAGE_UNSTABLE:" + String(dataQuality.voltageStability, 1) + "V";
    currentAlert = "UNSTBL";
    shouldAlert = true;
    buzzerTone = 1200; // Medium frequency for instability
  }
  
  else {
    currentAlert = "";
  }
  
  buzzerState = shouldAlert && buzzerEnabled;
  
  if (shouldAlert) {
    alertCount++;
    // Send alert to MQTT
    if (mqttClient.connected()) {
      mqttClient.publish(alert_feed, alertMsg.c_str());
    }
    Serial.println("ALERT #" + String(alertCount) + ": " + alertMsg);
  }
}

void handleBuzzer() {
  if (buzzerState && buzzerEnabled) {
    if (millis() - lastBuzzerToggle >= 500) { // Toggle every 500ms
      tone(BUZZER_PIN, buzzerTone, 200);
      lastBuzzerToggle = millis();
    }
  } else {
    noTone(BUZZER_PIN);
  }
}

void handleModeButton() {
  static bool lastButtonState = HIGH;
  static unsigned long lastDebounceTime = 0;
  const unsigned long debounceDelay = 50;
  
  bool reading = digitalRead(MODE_BUTTON_PIN);
  
  if (reading != lastButtonState) {
    lastDebounceTime = millis();
  }
  
  if ((millis() - lastDebounceTime) > debounceDelay) {
    if (reading == LOW && lastButtonState == HIGH) {
      // Button pressed - cycle display mode
      displayMode = (DisplayMode)((displayMode + 1) % 6);
      autoDisplayChange = false; // Disable auto mode switching
      Serial.print("Manual display mode: ");
      Serial.println(displayMode);
      lastDisplayChange = millis();
    }
  }
  
  lastButtonState = reading;
}

void handleSerialCommands() {
  if (Serial.available()) {
    String command = Serial.readStringUntil('\n');
    command.trim();
    command.toLowerCase();
    
    if (command.startsWith("threshold ")) {
      String param = command.substring(10);
      if (param.startsWith("lowv ")) {
        lowVoltageThreshold = param.substring(5).toFloat();
        Serial.print("Low voltage threshold set to: ");
        Serial.println(lowVoltageThreshold);
      } else if (param.startsWith("highv ")) {
        highVoltageThreshold = param.substring(6).toFloat();
        Serial.print("High voltage threshold set to: ");
        Serial.println(highVoltageThreshold);
      } else if (param.startsWith("highi ")) {
        highCurrentThreshold = param.substring(6).toFloat();
        Serial.print("High current threshold set to: ");
        Serial.println(highCurrentThreshold);
      } else if (param.startsWith("highp ")) {
        highPowerThreshold = param.substring(6).toFloat();
        Serial.print("High power threshold set to: ");
        Serial.println(highPowerThreshold);
      } else if (param.startsWith("phantom ")) {
        phantomThreshold = param.substring(8).toFloat();
        Serial.print("Phantom threshold set to: ");
        Serial.println(phantomThreshold);
      } else if (param.startsWith("pf ")) {
        lowPowerFactorThreshold = param.substring(3).toFloat();
        Serial.print("Low power factor threshold set to: ");
        Serial.println(lowPowerFactorThreshold);
      }
    }
    else if (command == "buzzer on") {
      buzzerEnabled = true;
      Serial.println("Buzzer enabled");
    }
    else if (command == "buzzer off") {
      buzzerEnabled = false;
      buzzerState = false;
      noTone(BUZZER_PIN);
      Serial.println("Buzzer disabled");
    }
    else if (command == "auto on") {
      autoDisplayChange = true;
      Serial.println("Auto display mode enabled");
    }
    else if (command == "auto off") {
      autoDisplayChange = false;
      Serial.println("Auto display mode disabled");
    }
    else if (command.startsWith("mode ")) {
      int newMode = command.substring(5).toInt();
      if (newMode >= 0 && newMode < 6) {
        displayMode = (DisplayMode)newMode;
        autoDisplayChange = false;
        Serial.print("Display mode set to: ");
        Serial.println(displayMode);
      } else {
        Serial.println("Invalid mode. Use 0-5.");
      }
    }
    else if (command == "reset stats") {
      resetStatistics();
      Serial.println("Statistics reset");
    }
    else if (command == "reset energy") {
      totalEnergyWh = 0.0;
      dailyEnergyWh = 0.0;
      Serial.println("Energy counters reset");
    }
    else if (command == "save") {
      saveSettings();
      Serial.println("Settings saved to flash");
    }
    else if (command == "load") {
      loadSettings();
      Serial.println("Settings loaded from flash");
    }
    else if (command == "status") {
      printSystemStatus();
    }
    else if (command == "help") {
      printHelp();
    }
    else if (command == "force11") {
      Serial.println("Forcing Channel 11...");
      forceChannel11();
    }
    else if (command == "diagnostics") {
      printNetworkDiagnostics();
    }
    else if (command.length() > 0) {
      Serial.println("Unknown command. Type 'help' for available commands.");
    }
  }
}

void printSystemStatus() {
  Serial.println("\n=== System Status ===");

// Callback function for ESP-NOW data reception
void onDataReceived(const esp_now_recv_info *recv_info, const uint8_t *incomingData, int len) {
  if (len == sizeof(SensorData)) {
    // Store previous data for comparison
    lastValidData = receivedData;
    
    // Copy new data
    memcpy(&receivedData, incomingData, sizeof(receivedData));
    newDataReceived = true;
    dataValid = true;
    lastDataReceived = millis();
    dataQuality.consecutiveTimeouts = 0; // Reset timeout counter
    
    // Update statistics
    updateStatistics();
    
    // Print detailed received data
    Serial.print("Data received from: ");
    for (int i = 0; i < 6; i++) {
      Serial.printf("%02X", recv_info->src_addr[i]);
      if (i < 5) Serial.print(":");
    }
    Serial.print(" | V: ");
    Serial.print(receivedData.liveVoltage, 1);
    Serial.print("V, I: ");
    Serial.print(receivedData.current, 3);
    Serial.print("A, P: ");
    Serial.print(receivedData.power, 1);
    Serial.print("W, PF: ");
    Serial.print(receivedData.powerFactor, 2);
    Serial.print(" | RSSI: ");
    Serial.println(receivedData.signalStrength);
    
    stats.successfulTransmissions++;
  } else {
    Serial.print("Invalid data size received: ");
    Serial.print(len);
    Serial.print(" bytes, expected: ");
    Serial.println(sizeof(SensorData));
    stats.failedTransmissions++;
  }
}

void forceChannel11() {
  Serial.println("\n=== FORCING CHANNEL 11 ===");
  
  // Complete WiFi shutdown
  WiFi.disconnect(true);
  WiFi.mode(WIFI_OFF);
  delay(1000);
  
  // Start in STA mode
  WiFi.mode(WIFI_STA);
  WiFi.setSleep(false);
  delay(500);
  
  // Multiple attempts to force channel 11
  for (int attempt = 1; attempt <= 5; attempt++) {
    Serial.print("Attempt ");
    Serial.print(attempt);
    Serial.print(" to force Channel 11... ");
    
    // Force channel using esp_wifi_set_channel
    esp_err_t result = esp_wifi_set_channel(11, WIFI_SECOND_CHAN_NONE);
    delay(1000);
    
    // Verify channel
    uint8_t primary;
    wifi_second_chan_t second;
    esp_wifi_get_channel(&primary, &second);
    
    Serial.print("Result: ");
    if (result == ESP_OK) {
      Serial.print("OK, ");
    } else {
      Serial.print("FAILED, ");
    }
    Serial.print("Channel: ");
    Serial.println(primary);
    
    if (primary == 11) {
      Serial.println("SUCCESS: Channel 11 locked!");
      wifiChannel = 11;
      channelLocked = true;
      break;
    }
    
    delay(500);
  }
  
  // If still not on channel 11, try alternative method
  if (WiFi.channel() != 11) {
    Serial.println("Using alternative channel forcing method...");
    
    WiFi.mode(WIFI_OFF);
    delay(1000);
    
    // Set WiFi config manually
    wifi_init_config_t cfg = WIFI_INIT_CONFIG_DEFAULT();
    esp_wifi_init(&cfg);
    esp_wifi_set_mode(WIFI_MODE_STA);
    esp_wifi_set_channel(11, WIFI_SECOND_CHAN_NONE);
    esp_wifi_start();
    
    delay(1000);
    
    // Verify final channel
    uint8_t finalChannel;
    wifi_second_chan_t second;
    esp_wifi_get_channel(&finalChannel, &second);
    
    Serial.print("Final forced channel: ");
    Serial.println(finalChannel);
    
    if (finalChannel == 11) {
      Serial.println("SUCCESS: Alternative method worked!");
      wifiChannel = 11;
      channelLocked = true;
    } else {
      Serial.println("CRITICAL: Cannot force Channel 11!");
    }
  }
  
  Serial.println("=== CHANNEL FORCING COMPLETE ===\n");
}

void connectToWiFiWithChannelLock() {
  lcd.clear();
  lcd.setCursor(0, 0);
  lcd.print("Connecting WiFi");
  lcd.setCursor(0, 1);
  lcd.print(ssid);
  
  Serial.println("\n=== WiFi Connection with Channel 11 Lock ===");
  
  // Ensure we're in AP_STA mode for ESP-NOW + WiFi
  WiFi.mode(WIFI_AP_STA);
  delay(1000);
  
  // Force channel 11 again
  esp_wifi_set_channel(11, WIFI_SECOND_CHAN_NONE);
  delay(500);
  
  // Scan for networks
  Serial.println("Scanning for networks...");
  int networks = WiFi.scanNetworks(false, true);
  
  bool targetFound = false;
  int targetChannel = 11;
  
  for (int i = 0; i < networks; i++) {
    Serial.print("Network: ");
    Serial.print(WiFi.SSID(i));
    Serial.print(" Ch:");
    Serial.print(WiFi.channel(i));
    Serial.print(" RSSI:");
    Serial.println(WiFi.RSSI(i));
    
    if (WiFi.SSID(i) == ssid) {
      targetFound = true;
      targetChannel = WiFi.channel(i);
      Serial.print("*** TARGET FOUND on Channel ");
      Serial.println(targetChannel);
      
      if (targetChannel != 11) {
        Serial.println("WARNING: Target network is NOT on Channel 11!");
        Serial.println("ESP-NOW communication may not work properly!");
        Serial.println("Please change your hotspot to Channel 11");
        
        lcd.clear();
        lcd.setCursor(0, 0);
        lcd.print("WRONG CHANNEL!");
        lcd.setCursor(0, 1);
        lcd.print("Hotspot Ch:");
        lcd.print(targetChannel);
        lcd.setCursor(0, 2);
        lcd.print("Need Ch:11");
        lcd.setCursor(0, 3);
        lcd.print("Fix hotspot!");
        delay(5000);
      }
    }
  }
  
  if (!targetFound) {
    Serial.println("Target network not found!");
    lcd.clear();
    lcd.setCursor(0, 0);
    lcd.print("Network not found");
    lcd.setCursor(0, 1);
    lcd.print("Check hotspot");
    
    // Still maintain channel 11 for ESP-NOW
    esp_wifi_set_channel(11, WIFI_SECOND_CHAN_NONE);
    delay(3000);
    return;
  }
  
  // Multiple connection attempts with channel forcing
  for (int connAttempt = 1; connAttempt <= 3; connAttempt++) {
    Serial.print("Connection attempt ");
    Serial.print(connAttempt);
    Serial.println("/3");
    
    // Force channel 11 before each attempt
    esp_wifi_set_channel(11, WIFI_SECOND_CHAN_NONE);
    delay(1000);
    
    WiFi.begin(ssid, password);
    
    int attempts = 0;
    while (WiFi.status() != WL_CONNECTED && attempts < 30) {
      delay(500);
      Serial.print(".");
      attempts++;
      
      if (attempts % 10 == 0) {
        Serial.print(" [");
        Serial.print(attempts);
        Serial.print("/30] ");
        
        // Re-force channel 11 every 10 attempts
        esp_wifi_set_channel(11, WIFI_SECOND_CHAN_NONE);
      }
      
      lcd.setCursor(0, 2);
      lcd.print("Attempt: ");
      lcd.print(attempts);
      lcd.print("/30");
    }
    
    if (WiFi.status() == WL_CONNECTED) {
      break;
    } else {
      Serial.println(" FAILED");
      WiFi.disconnect();
      delay(2000);
    }
  }
  
  if (WiFi.status() == WL_CONNECTED) {
    Serial.println("\nWiFi Connected!");
    Serial.print("IP Address: ");
    Serial.println(WiFi.localIP());
    
    int actualChannel = WiFi.channel();
    Serial.print("WiFi Channel: ");
    Serial.println(actualChannel);
    
    if (actualChannel != 11) {
      Serial.println("*** CRITICAL: WiFi connected on wrong channel! ***");
      Serial.println("Forcing back to Channel 11...");
      
      // Try to force back to channel 11
      for (int i = 0; i < 3; i++) {
        esp_wifi_set_channel(11, WIFI_SECOND_CHAN_NONE);
        delay(1000);
        if (WiFi.channel() == 11) {
          Serial.println("Successfully restored Channel 11!");
          break;
        }
      }
      
      if (WiFi.channel() != 11) {
        Serial.println("FAILED to restore Channel 11!");
        Serial.println("ESP-NOW may not work properly!");
      }
    }
    
    lcd.clear();
    lcd.setCursor(0, 0);
    lcd.print("WiFi Connected!");
    lcd.setCursor(0, 1);
    lcd.print("Ch:");
    lcd.print(WiFi.channel());
    if (WiFi.channel() == 11) {
      lcd.print(" - CORRECT");
    } else {
      lcd.print(" - WRONG!");
    }
    lcd.setCursor(0, 2);
    lcd.print("IP:");
    lcd.print(WiFi.localIP().toString().substring(0, 12));
    
    // Run network diagnostics
    printNetworkDiagnostics();
    
  } else {
    Serial.println("WiFi connection failed!");
    
    lcd.clear();
    lcd.setCursor(0, 0);
    lcd.print("WiFi Failed!");
    lcd.setCursor(0, 1);
    lcd.print("ESP-NOW only");
    
    // Ensure ESP-NOW still works on channel 11
    esp_wifi_set_channel(11, WIFI_SECOND_CHAN_NONE);
  }
  
  delay(3000);
  Serial.println("=== WiFi Setup Complete ===\n");
}

  delay(500);

  delay(1000);
  
  WiFi.begin(ssid, password);
  
  int attempts = 0;
  while (WiFi.status() != WL_CONNECTED && attempts < 20) {

      esp_wifi_set_channel(11, WIFI_SECOND_CHAN_NONE);

  } else {

  }
}

void loop() {
  // Handle serial commands
  handleSerialCommands();
  
  // Check manual mode button
  handleModeButton();
  
  // Verify channel 11 every 30 seconds
  static unsigned long lastChannelCheck = 0;
  if (millis() - lastChannelCheck > 30000) {
    if (WiFi.channel() != 11) {
      Serial.println("Channel drift detected! Forcing back to 11...");
      esp_wifi_set_channel(11, WIFI_SECOND_CHAN_NONE);
    }
    lastChannelCheck = millis();
  }
  
  // Check WiFi status and reconnect if needed
  if (WiFi.status() != WL_CONNECTED && millis() > 60000) { // Wait 1 minute after boot
    static unsigned long lastReconnectAttempt = 0;
    if (millis() - lastReconnectAttempt > 30000) { // Try every 30 seconds
      Serial.println("WiFi disconnected, attempting reconnection...");
      reconnectWiFiSameChannel();
      lastReconnectAttempt = millis();
    }
  }
  
  // Check for data timeout
  if (dataValid && (millis() - lastDataReceived > DATA_TIMEOUT)) {
    dataValid = false;
    dataQuality.consecutiveTimeouts++;
    dataQuality.totalTimeouts++;
    Serial.println("WARNING: Data timeout - no signal from sender");
  }
  
  // Process new data if received
  if (newDataReceived && dataValid) {
    newDataReceived = false;
    
    // Update energy calculation
    updateEnergy();
    
    // Check for alerts
    checkAlerts();
    
    // Update data quality metrics
    updateDataQuality();
  }
  
  // Send data to MQTT periodically
  if (millis() - lastMqttSend >= MQTT_SEND_INTERVAL && dataValid) {
    sendToMQTT();
    lastMqttSend = millis();
  }
  
  // Handle MQTT connection
  if (!mqttClient.connected()) {
    reconnectMQTT();
  }
  mqttClient.loop();
  
  // Handle buzzer alerts
  handleBuzzer();
  
  // Update LCD with rotating display or manual mode
  if (autoDisplayChange && (millis() - lastDisplayChange > DISPLAY_CHANGE_INTERVAL)) {
    displayMode = (DisplayMode)((displayMode + 1) % 6);
    lastDisplayChange = millis();
  }
  updateLCD();
  
  // Reset daily energy at midnight (24 hours)
  if (millis() - dayStartTime > DAY_RESET_INTERVAL) {
    dailyEnergyWh = 0.0;
    dayStartTime = millis();
    Serial.println("Daily energy counter reset");
  }
  
  // Periodic status update
  if (millis() - lastStatusUpdate > STATUS_UPDATE_INTERVAL) {
    lastStatusUpdate = millis();
    
    // Debug output every 30 seconds
    static unsigned long lastDebugOutput = 0;
    if (millis() - lastDebugOutput > 30000) {
      printSystemStatus();
      lastDebugOutput = millis();
    }
  }
  
  delay(50); // Reduced delay for more responsive system
}

void updateStatistics() {
  if (!dataValid) return;
  
  stats.dataPoints++;
  
  // Update min/max values
  if (receivedData.liveVoltage > stats.maxVoltage) stats.maxVoltage = receivedData.liveVoltage;
  if (receivedData.liveVoltage < stats.minVoltage && receivedData.liveVoltage > 10) stats.minVoltage = receivedData.liveVoltage;
  if (receivedData.current > stats.maxCurrent) stats.maxCurrent = receivedData.current;
  if (receivedData.power > stats.maxPower) stats.maxPower = receivedData.power;
  
  // Update running averages
  stats.avgVoltage = ((stats.avgVoltage * (stats.dataPoints - 1)) + receivedData.liveVoltage) / stats.dataPoints;
  stats.avgCurrent = ((stats.avgCurrent * (stats.dataPoints - 1)) + receivedData.current) / stats.dataPoints;
  stats.avgPower = ((stats.avgPower * (stats.dataPoints - 1)) + receivedData.power) / stats.dataPoints;
  
  // Update signal strength average
  dataQuality.averageSignalStrength = ((dataQuality.averageSignalStrength * (stats.dataPoints - 1)) + receivedData.signalStrength) / stats.dataPoints;
}

void updateDataQuality() {
  // Calculate voltage stability (standard deviation over recent readings)
  static float voltageHistory[10];
  static int voltageIndex = 0;
  static bool historyFilled = false;
  
  voltageHistory[voltageIndex] = receivedData.liveVoltage;
  voltageIndex = (voltageIndex + 1) % 10;
  if (voltageIndex == 0) historyFilled = true;
  
  if (historyFilled) {
    float avgVolt = 0;
    for (int i = 0; i < 10; i++) avgVolt += voltageHistory[i];
    avgVolt /= 10;
    
    float variance = 0;
    for (int i = 0; i < 10; i++) {
      float diff = voltageHistory[i] - avgVolt;
      variance += diff * diff;
    }
    dataQuality.voltageStability = sqrt(variance / 10);
  }
}

void reconnectMQTT() {
  static unsigned long lastAttempt = 0;
  if (millis() - lastAttempt < 5000) return;
  
  Serial.print("Connecting to MQTT...");
  
  String clientId = "ESP32VoltCurrentRx_" + WiFi.macAddress();
  clientId.replace(":", "");
  
  if (mqttClient.connect(clientId.c_str(), mqtt_username, mqtt_password)) {
    Serial.println(" connected");
  } else {
    Serial.print(" failed, rc=");
    Serial.print(mqttClient.state());
    Serial.println(" trying again in 5 seconds");
  }
  
  lastAttempt = millis();
}

void updateEnergy() {
  if (millis() - lastEnergyUpdate >= ENERGY_UPDATE_INTERVAL) {
    if (receivedData.power > 0) {
      // Calculate energy consumed in the last interval (Wh = W * hours)
      float intervalHours = (float)(millis() - lastEnergyUpdate) / 3600000.0;
      float energyIncrement = receivedData.power * intervalHours;
      
      totalEnergyWh += energyIncrement;
      dailyEnergyWh += energyIncrement;
      
      Serial.print("Energy update: +");
      Serial.print(energyIncrement, 4);
      Serial.print("Wh, Total: ");
      Serial.print(totalEnergyWh, 2);
      Serial.println("Wh");
    }
